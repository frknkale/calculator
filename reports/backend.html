package main

import (
        "context"
        "errors"
        "log"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculator/backend/internal/httpapi"
)

func main() {
        addr := os.Getenv("ADDR")
        if addr == "" {
                addr = ":8080"
        }
        var frontend http.Handler
        if directory := os.Getenv("STATIC_DIR"); directory != "" {
                frontend = http.FileServer(http.Dir(directory))
        }
        server := &http.Server{
                Addr:              addr,
                Handler:           httpapi.NewHandler(frontend),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        shutdownDone := make(chan struct{})
        go func() {
                defer close(shutdownDone)
                <-ctx.Done()
                shutdown, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                defer cancel()
                if err := server.Shutdown(shutdown); err != nil {
                        log.Printf("shutdown: %v", err)
                }
        }()
        log.Printf("Calculator listening on %s", addr)
        if err := server.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                log.Fatal(err)
        }
        // Wait for active requests to drain before the main goroutine exits.
        <-shutdownDone
}

// Package calculator contains arithmetic rules independent of HTTP or UI code.
package calculator

import (
        "errors"
        "math"
)

var (
        ErrOperation      = errors.New("Choose a supported operation.")
        ErrOperand        = errors.New("Provide the required finite numeric operands.")
        ErrDivisionByZero = errors.New("Cannot divide by zero.")
        ErrNegativeRoot   = errors.New("Cannot take the square root of a negative number.")
        ErrResult         = errors.New("The result is outside the supported real-number range.")
)

// Calculate applies the selected operation. Square root uses only a.
// Percentage means b percent of a. Power follows math.Pow, including 0^0 = 1.
func Calculate(operation string, a, b *float64) (float64, error) {
        binary := false
        switch operation {
        case "add", "subtract", "multiply", "divide", "power", "percentage":
                binary = true
        case "sqrt":
        default:
                return 0, ErrOperation
        }
        if !finite(a) || (binary && !finite(b)) || (!binary && b != nil) {
                return 0, ErrOperand
        }

        var result float64
        switch operation {
        case "add":
                result = *a + *b
        case "subtract":
                result = *a - *b
        case "multiply":
                result = *a * *b
        case "divide":
                if *b == 0 {
                        return 0, ErrDivisionByZero
                }
                result = *a / *b
        case "power":
                result = math.Pow(*a, *b)
        case "sqrt":
                if *a < 0 {
                        return 0, ErrNegativeRoot
                }
                result = math.Sqrt(*a)
        case "percentage":
                result = *a * (*b / 100)
        }
        if math.IsNaN(result) || math.IsInf(result, 0) {
                return 0, ErrResult
        }
        return result, nil
}

func finite(value *float64) bool {
        return value != nil && !math.IsNaN(*value) && !math.IsInf(*value, 0)
}

// Package httpapi translates HTTP requests into calculator calls.
package httpapi

import (
        "encoding/json"
        "errors"
        "io"
        "mime"
        "net/http"

        "calculator/backend/internal/calculator"
)

type request struct {
        Operation string   `json:"operation"`
        A         *float64 `json:"a"`
        B         *float64 `json:"b,omitempty"`
}

type apiError struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// NewHandler returns the API routes and optional frontend handler.
func NewHandler(frontend http.Handler) http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("/api/calculate", calculate)
        mux.HandleFunc("/api/health", func(w http.ResponseWriter, r *http.Request) {
                if r.Method != http.MethodGet {
                        w.Header().Set("Allow", http.MethodGet)
                        writeError(w, http.StatusMethodNotAllowed, "method_not_allowed", "Use GET for this endpoint.")
                        return
                }
                writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
        })
        mux.HandleFunc("/api/", func(w http.ResponseWriter, r *http.Request) {
                writeError(w, http.StatusNotFound, "not_found", "API endpoint not found.")
        })
        if frontend != nil {
                mux.Handle("/", frontend)
        }
        return mux
}

func calculate(w http.ResponseWriter, r *http.Request) {
        if r.Method != http.MethodPost {
                w.Header().Set("Allow", http.MethodPost)
                writeError(w, http.StatusMethodNotAllowed, "method_not_allowed", "Use POST for this endpoint.")
                return
        }
        mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
        if err != nil || mediaType != "application/json" {
                writeError(w, http.StatusUnsupportedMediaType, "unsupported_media_type", "Send Content-Type: application/json.")
                return
        }
        r.Body = http.MaxBytesReader(w, r.Body, 4096)
        decoder := json.NewDecoder(r.Body)
        decoder.DisallowUnknownFields()
        var input request
        if err := decoder.Decode(&input); err != nil {
                invalidJSON(w, err)
                return
        }
        // A second value (or malformed trailing data) must not be silently ignored.
        if err := decoder.Decode(&struct{}{}); err != io.EOF {
                invalidJSON(w, err)
                return
        }
        result, err := calculator.Calculate(input.Operation, input.A, input.B)
        if err != nil {
                writeError(w, http.StatusBadRequest, "invalid_calculation", err.Error())
                return
        }
        writeJSON(w, http.StatusOK, map[string]float64{"result": result})
}

func invalidJSON(w http.ResponseWriter, err error) {
        var tooLarge *http.MaxBytesError
        if errors.As(err, &tooLarge) {
                writeError(w, http.StatusRequestEntityTooLarge, "body_too_large", "Request body must not exceed 4096 bytes.")
                return
        }
        writeError(w, http.StatusBadRequest, "invalid_json", "Send one JSON object with operation and numeric a/b fields only.")
}

func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, struct {
                Error apiError `json:"error"`
        }{Error: apiError{Code: code, Message: message}})
}

func writeJSON(w http.ResponseWriter, status int, value any) {
        w.Header().Set("Content-Type", "application/json")
        w.Header().Set("X-Content-Type-Options", "nosniff")
        w.WriteHeader(status)
        // Response values contain only strings and finite numbers.
        _ = json.NewEncoder(w).Encode(value)
}
